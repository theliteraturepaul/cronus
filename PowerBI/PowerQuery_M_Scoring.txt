// ============================================================================
// QUERY 2 of 5: ActiveDataset
// Load: enabled. Returns one row: the dual-grain summary of the active dataset.
// ============================================================================
let
    Source = PostgreSQL.Database(Config[PG_Server], Config[PG_Database]),
    Data = Value.NativeQuery(
        Source,
        "
        SELECT
            d.id::text AS dataset_id,
            d.name,
            d.as_of_date,
            agg.relationship_count,
            agg.unique_customers,
            agg.relationship_churn_count,
            ROUND(100.0 * agg.relationship_churn_count / NULLIF(agg.relationship_count, 0), 2)
                AS relationship_churn_rate,
            agg.current_relationship_count,
            agg.current_churn_count,
            ROUND(100.0 * agg.current_churn_count / NULLIF(agg.current_relationship_count, 0), 2)
                AS current_churn_rate,
            cust.multi_relationship_customers,
            reentry.reentry_pairs,
            agg.partner_count,
            agg.min_registration_date,
            agg.max_registration_date
        FROM datasets AS d
        CROSS JOIN LATERAL (
            SELECT
                COUNT(*) AS relationship_count,
                COUNT(DISTINCT r.customer_id) AS unique_customers,
                COUNT(*) FILTER (WHERE r.churn) AS relationship_churn_count,
                COUNT(*) FILTER (WHERE r.is_latest_relationship) AS current_relationship_count,
                COUNT(*) FILTER (WHERE r.is_latest_relationship AND r.churn) AS current_churn_count,
                COUNT(DISTINCT r.telecom_partner) AS partner_count,
                MIN(r.registration_date) AS min_registration_date,
                MAX(r.registration_date) AS max_registration_date
            FROM customer_relationships AS r
            WHERE r.dataset_id = d.id
        ) AS agg
        CROSS JOIN LATERAL (
            SELECT COUNT(*) AS multi_relationship_customers
            FROM (
                SELECT r.customer_id
                FROM customer_relationships AS r
                WHERE r.dataset_id = d.id
                GROUP BY r.customer_id
                HAVING COUNT(*) > 1
            ) AS multi
        ) AS cust
        CROSS JOIN LATERAL (
            SELECT COUNT(*) AS reentry_pairs
            FROM (
                SELECT r.customer_id, r.telecom_partner
                FROM customer_relationships AS r
                WHERE r.dataset_id = d.id
                GROUP BY r.customer_id, r.telecom_partner
                HAVING COUNT(*) > 1
            ) AS pairs
        ) AS reentry
        WHERE d.is_active = TRUE
        LIMIT 1
        ",
        null,
        [EnableFolding = false]
    ),
    Typed = Table.TransformColumnTypes(
        Data,
        {
            {"dataset_id", type text},
            {"name", type text},
            {"as_of_date", type date},
            {"relationship_count", Int64.Type},
            {"unique_customers", Int64.Type},
            {"relationship_churn_count", Int64.Type},

        }
    )
in
    Typed

// ============================================================================
// QUERY 4 of 5: DateTable
// Load: enabled. Returns one row per date from the first registration year to
// the end of the snapshot year.
// ============================================================================
let
    Snapshot = Date.From(ActiveDataset{0}[as_of_date]),
    FirstRegistration = Date.From(ActiveDataset{0}[min_registration_date]),
    StartDate = Date.StartOfYear(FirstRegistration),
    EndDate = Date.EndOfYear(Snapshot),
    DateList = List.Dates(
        StartDate,
        Duration.Days(EndDate - StartDate) + 1,
        #duration(1, 0, 0, 0)
    ),
    DatesTable = Table.FromList(DateList, Splitter.SplitByNothing(), {"Date"}),
    Typed = Table.TransformColumnTypes(DatesTable, {{"Date", type date}}),
    WithYear = Table.AddColumn(Typed, "Year", each Date.Year([Date]), Int64.Type),
    WithQuarter = Table.AddColumn(
        WithYear,
        "Quarter",
        each "Q" & Text.From(Date.QuarterOfYear([Date])),
        type text
    ),
    WithMonth = Table.AddColumn(WithQuarter, "Month", each Date.Month([Date]), Int64.Type),
    WithMonthName = Table.AddColumn(
        WithMonth,
        "Month Name",
        each Date.ToText([Date], "MMM"),
        type text
    ),
    WithWeek = Table.AddColumn(
        WithMonthName,
        "Week",
        each Date.WeekOfYear([Date]),
        Int64.Type
    )
in
    WithWeek


// ============================================================================
// QUERY 5 of 5: ScoredRelationships
// Load: enabled. Returns the current relationships with a churn probability,
// both fixed-threshold flags, both priorities, both thresholds, and a rank.
// ============================================================================
let
    Coefficient = (name as text) as number =>
        if Record.HasFields(Config[Coefficients], {name})
        then Record.Field(Config[Coefficients], name)
        else 0,

    CategoricalScore = (prefix as text, value as nullable text, known as list) as number =>
        if value <> null and List.Contains(known, value)
        then Coefficient(prefix & value)
        else 0,

    StateScore = (value as nullable text) as number =>
        if value = null then 0
        else if List.Contains(Config[Cat_StateFrequent], value)
            then Coefficient("state_" & value)
        else if List.Contains(Config[Cat_StateInfrequent], value)
            then Coefficient("state_infrequent_sklearn")
        else 0,

    NumericScore = (value as number, feature as text) as number =>
        (Number.From(value) - Record.Field(Config[Num_Mean], feature))
            / Record.Field(Config[Num_Scale], feature)
            * Coefficient(feature),

    LinearScore = (row as record) as number =>
        Config[Model_Intercept]
        + CategoricalScore(
            "telecom_partner_",
            row[telecom_partner],
            Config[Cat_TelecomPartner]
        )
        + CategoricalScore("gender_", row[gender], Config[Cat_Gender])
        + StateScore(row[state])
        + NumericScore(row[age], "age")
        + NumericScore(row[num_dependents], "num_dependents")
        + NumericScore(row[estimated_salary], "estimated_salary")
        + NumericScore(row[calls_made], "calls_made")
        + NumericScore(row[sms_sent], "sms_sent")
        + NumericScore(row[data_used], "data_used")
        + NumericScore(row[tenure_days], "tenure_days"),

    Sigmoid = (z as number) as number =>
        if z >= 0
        then 1 / (1 + Number.Exp(-z))
        else
            let
                e = Number.Exp(z)
            in
                e / (1 + e),

    AddProbability = Table.AddColumn(
        CurrentRelationships,
        "churn_probability",
        each Sigmoid(LinearScore(_)),
        type number
    ),
    AddDefaultFlag = Table.AddColumn(
        AddProbability,
        "predicted_churn_default",
        each [churn_probability] >= Config[Threshold_Default],
        type logical
    ),
    AddRecall70Flag = Table.AddColumn(
        AddDefaultFlag,
        "predicted_churn_recall70",
        each [churn_probability] >= Config[Threshold_Recall70],
        type logical
    ),
    AddDefaultPriority = Table.AddColumn(
        AddRecall70Flag,
        "review_priority_default",
        each
            if [predicted_churn_default] then "High"
            else if [churn_probability] >= Config[Threshold_Default] / 2 then "Elevated"
            else "Low",
        type text
    ),
    AddRecall70Priority = Table.AddColumn(
        AddDefaultPriority,
        "review_priority_recall70",
        each
            if [predicted_churn_recall70] then "High"
            else if [churn_probability] >= Config[Threshold_Recall70] / 2 then "Elevated"
            else "Low",
        type text
    ),
    AddDefaultThreshold = Table.AddColumn(
        AddRecall70Priority,
        "threshold_default",
        each Config[Threshold_Default],
        type number
    ),
    AddRecall70Threshold = Table.AddColumn(
        AddDefaultThreshold,
        "threshold_recall70",
        each Config[Threshold_Recall70],
        type number
    ),
    Sorted = Table.Sort(
        AddRecall70Threshold,
        {{"churn_probability", Order.Descending}, {"customer_id", Order.Ascending}}
    ),
    Ranked = Table.AddIndexColumn(Sorted, "score_rank", 1, 1, Int64.Type),
    Final = Table.SelectColumns(
        Ranked,
        {
            "score_rank",
            "customer_id",
            "telecom_partner",
            "gender",
            "state",
            "age",
            "num_dependents",
            "estimated_salary",
            "calls_made",
            "sms_sent",
            "data_used",
            "tenure_days",
            "relationship_count",
            "date_of_registration",
            "churn",
            "churn_probability",
            "predicted_churn_default",
            "predicted_churn_recall70",
            "review_priority_default",
            "review_priority_recall70",
            "threshold_default",
            "threshold_recall70"
        }
    )
in
    Final
