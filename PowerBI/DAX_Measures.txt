// ============================================================================
// 3. CURRENT CUSTOMER CHURN (ScoredRelationships, latest rows only)
// ============================================================================

Provider N = COUNTROWS ( ScoredRelationships )

Provider Churned =
CALCULATE ( COUNTROWS ( ScoredRelationships ), ScoredRelationships[churn] = TRUE () )

Provider Churn Rate = DIVIDE ( [Provider Churned], [Provider N] )

Provider Share of Base =
DIVIDE ( [Provider N], CALCULATE ( [Provider N], ALL ( ScoredRelationships[telecom_partner] ) ) )

Provider Small Sample = IF ( [Provider N] < 100, "Small sample", "" )

Wilson Lower =
VAR p = [Provider Churn Rate]
VAR n = [Provider N]
VAR z = 1.96
VAR denominator = 1 + z ^ 2 / n
VAR centre = ( p + z ^ 2 / ( 2 * n ) ) / denominator
VAR margin =
    z * SQRT ( ( p * ( 1 - p ) / n ) + z ^ 2 / ( 4 * n ^ 2 ) ) / denominator
RETURN
    MAX ( 0, centre - margin )

Wilson Upper =
VAR p = [Provider Churn Rate]
VAR n = [Provider N]
VAR z = 1.96
VAR denominator = 1 + z ^ 2 / n
VAR centre = ( p + z ^ 2 / ( 2 * n ) ) / denominator
VAR margin =
    z * SQRT ( ( p * ( 1 - p ) / n ) + z ^ 2 / ( 4 * n ^ 2 ) ) / denominator
RETURN
    MIN ( 1, centre + margin )
